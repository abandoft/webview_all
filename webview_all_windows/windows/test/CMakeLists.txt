cmake_minimum_required(VERSION 3.15)
project(webview_all_windows_native_tests LANGUAGES CXX)

enable_testing()
add_executable(environment_request_test environment_request_test.cc)
target_compile_features(environment_request_test PRIVATE cxx_std_17)
target_include_directories(environment_request_test PRIVATE ../src)
if(MSVC)
  target_compile_options(environment_request_test PRIVATE /W4 /WX)
else()
  target_compile_options(environment_request_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME environment_request_test COMMAND environment_request_test)

add_executable(surface_bounds_test surface_bounds_test.cc)
target_compile_features(surface_bounds_test PRIVATE cxx_std_17)
target_include_directories(surface_bounds_test PRIVATE ../src)
if(MSVC)
  target_compile_options(surface_bounds_test PRIVATE /W4 /WX)
else()
  target_compile_options(surface_bounds_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME surface_bounds_test COMMAND surface_bounds_test)

if(WIN32)
  add_executable(profile_name_test profile_name_test.cc)
  target_compile_features(profile_name_test PRIVATE cxx_std_17)
  target_include_directories(profile_name_test PRIVATE ../src)
  target_link_libraries(profile_name_test PRIVATE bcrypt)
  if(MSVC)
    target_compile_options(profile_name_test PRIVATE /W4 /WX)
  endif()
  add_test(NAME profile_name_test COMMAND profile_name_test)
endif()
